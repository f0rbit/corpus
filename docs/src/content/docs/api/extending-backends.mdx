import { Aside } from '@astrojs/starlight/components';

If the built-in backends (memory, file, Cloudflare, remote) don't fit your storage, write your own. A backend is a plain object that satisfies the exported `Backend` type. There is no base class and no registration step: anything that type-checks as a `Backend` can go into `create_corpus().with_backend(...)`.

This page uses only names exported from `@f0rbit/corpus` and `@f0rbit/corpus/testing`.

<Aside type="note">
  The built-in backends are assembled from internal adapter helpers in corpus's `backend/` directory. Those helpers are not part of the public API and may change between releases. Custom backends implement `MetadataClient` and `DataClient` directly, as shown below.
</Aside>

## The `Backend` type

```typescript
import type { BatchOp, CorpusError, DataClient, EventHandler, MetadataClient, ObservationsClient, Result } from '@f0rbit/corpus'

type Backend = {
  metadata: MetadataClient
  data: DataClient
  observations?: ObservationsClient
  on_event?: EventHandler
  apply_batch?: (ops: BatchOp[]) => Promise<Result<void, CorpusError>>
}
```

| Member         | Required | What it does                                                                                              |
| -------------- | -------- | --------------------------------------------------------------------------------------------------------- |
| `metadata`     | yes      | Snapshot metadata: get/put/delete, filtered listing, `get_latest`, lineage children, hash lookup           |
| `data`         | yes      | Content-addressed blobs keyed by `data_key`                                                               |
| `observations` | no       | Enables `corpus.observations`. Without it, observation APIs are unavailable                               |
| `on_event`     | no       | The handler the backend reports `CorpusEvent`s to, `error` events included                                |
| `apply_batch`  | no       | Atomic commit for `corpus.transaction()`. Without it, transactions use a sequential best-effort fallback  |

See [Backend Types](/corpus/api/backends/types) for the full `MetadataClient` and `DataClient` signatures.

## Implementing `Backend` directly

The example below is a complete backend built on `Map`s. Swap the maps for your storage (a KV namespace, a SQL table, an object store) and keep the same `Result` shapes and ordering rules.

```typescript
import { create_observations_client, create_observations_storage, err, ok, try_catch_async } from '@f0rbit/corpus'
import type {
  Backend,
  BatchOp,
  CorpusError,
  CorpusEvent,
  DataClient,
  DataHandle,
  EventHandler,
  ListOpts,
  MetadataClient,
  ObservationRow,
  Result,
  SnapshotMeta,
} from '@f0rbit/corpus'

export type MapBackendOptions = { on_event?: EventHandler }

const meta_key = (store_id: string, version: string): string => `${store_id}\u0000${version}`

const as_error = (cause: unknown): Error => (cause instanceof Error ? cause : new Error(String(cause)))

const one_chunk = (bytes: Uint8Array): ReadableStream<Uint8Array> =>
  new ReadableStream({
    start(controller) {
      controller.enqueue(bytes)
      controller.close()
    },
  })

const read_all = async (input: ReadableStream<Uint8Array> | Uint8Array): Promise<Uint8Array> =>
  input instanceof Uint8Array ? input : new Uint8Array(await new Response(input).arrayBuffer())

// Newest first. Map iteration is insertion order, so reversing before the
// stable sort means the most recently put snapshot wins a created_at tie.
const newest_first = (metas: SnapshotMeta[]): SnapshotMeta[] =>
  metas.toReversed().toSorted((a, b) => b.created_at.getTime() - a.created_at.getTime())

const matches = (meta: SnapshotMeta, opts: ListOpts): boolean =>
  (opts.before === undefined || meta.created_at < opts.before) &&
  (opts.after === undefined || meta.created_at > opts.after) &&
  (opts.tags === undefined || opts.tags.every((tag) => meta.tags?.includes(tag) === true)) &&
  (opts.entity_key === undefined || meta.entity_key === opts.entity_key)

export function create_map_backend(options: MapBackendOptions = {}): Backend {
  const metas = new Map<string, SnapshotMeta>()
  const blobs = new Map<string, Uint8Array>()
  const rows = new Map<string, ObservationRow>()
  const emit = (event: CorpusEvent): void => options.on_event?.(event)

  const in_store = (store_id: string): SnapshotMeta[] => [...metas.values()].filter((m) => m.store_id === store_id)

  const metadata: MetadataClient = {
    async get(store_id, version): Promise<Result<SnapshotMeta, CorpusError>> {
      const meta = metas.get(meta_key(store_id, version))
      emit({ type: 'meta_get', store_id, version, found: meta !== undefined })
      if (meta === undefined) return err({ kind: 'not_found', store_id, version })
      return ok(meta)
    },

    async put(meta): Promise<Result<void, CorpusError>> {
      metas.set(meta_key(meta.store_id, meta.version), meta)
      emit({ type: 'meta_put', store_id: meta.store_id, version: meta.version })
      return ok(undefined)
    },

    async delete(store_id, version): Promise<Result<void, CorpusError>> {
      metas.delete(meta_key(store_id, version))
      emit({ type: 'meta_delete', store_id, version })
      return ok(undefined)
    },

    async *list(store_id, opts: ListOpts = {}) {
      // filter first, then limit
      const filtered = newest_first(in_store(store_id)).filter((m) => matches(m, opts))
      const page = opts.limit ? filtered.slice(0, opts.limit) : filtered
      yield* page
      emit({ type: 'meta_list', store_id, count: page.length })
    },

    async get_latest(store_id, opts = {}): Promise<Result<SnapshotMeta, CorpusError>> {
      const scoped = in_store(store_id).filter((m) => opts.entity_key === undefined || m.entity_key === opts.entity_key)
      const latest = newest_first(scoped).at(0)
      if (latest === undefined) return err({ kind: 'not_found', store_id, version: 'latest' })
      return ok(latest)
    },

    async *get_children(parent_store_id, parent_version) {
      yield* [...metas.values()].filter((m) =>
        m.parents.some((p) => p.store_id === parent_store_id && p.version === parent_version),
      )
    },

    async find_by_hash(store_id, content_hash) {
      return in_store(store_id).find((m) => m.content_hash === content_hash) ?? null
    },

    async *list_stores() {
      yield* [...new Set([...metas.values()].map((m) => m.store_id))].toSorted()
    },
  }

  const data: DataClient = {
    async get(data_key): Promise<Result<DataHandle, CorpusError>> {
      const bytes = blobs.get(data_key)
      if (bytes === undefined) return err({ kind: 'not_found', store_id: data_key, version: '' })
      return ok({ bytes: async () => bytes, stream: () => one_chunk(bytes) })
    },

    async put(data_key, input): Promise<Result<void, CorpusError>> {
      const read = await try_catch_async(
        () => read_all(input),
        (cause): CorpusError => ({ kind: 'storage_error', operation: 'data.put', cause: as_error(cause) }),
      )
      if (!read.ok) {
        emit({ type: 'error', error: read.error })
        return read
      }
      blobs.set(data_key, read.value)
      return ok(undefined)
    },

    async delete(data_key): Promise<Result<void, CorpusError>> {
      blobs.delete(data_key)
      return ok(undefined)
    },

    async exists(data_key) {
      return blobs.has(data_key)
    },
  }

  const observations = create_observations_client(
    create_observations_storage({
      get_all: async () => [...rows.values()],
      set_all: async (next) => {
        rows.clear()
        for (const row of next) rows.set(row.id, row)
      },
      get_one: async (id) => rows.get(id) ?? null,
      add_one: async (row): Promise<Result<void, CorpusError>> => {
        if (rows.has(row.id)) {
          const cause = new Error(`observation id "${row.id}" already exists`)
          return err({ kind: 'validation_error', message: cause.message, cause })
        }
        rows.set(row.id, row)
        return ok(undefined)
      },
      remove_one: async (id): Promise<Result<boolean, CorpusError>> => ok(rows.delete(id)),
    }),
    metadata,
    emit,
  )

  async function apply_batch(ops: BatchOp[]): Promise<Result<void, CorpusError>> {
    const ids = ops.flatMap((op) => (op.type === 'observation_put' ? [op.row.id] : []))
    const duplicate = ids.find((id, i) => rows.has(id) || ids.indexOf(id) !== i)
    if (duplicate !== undefined) {
      return err({
        kind: 'transaction_aborted',
        reason: 'apply_batch_failed',
        cause: new Error(`observation id "${duplicate}" already exists`),
      })
    }
    // Validation happened up front, and nothing below awaits, so no other
    // call can observe a half-applied batch.
    for (const op of ops) {
      switch (op.type) {
        case 'meta_put':
          metas.set(meta_key(op.meta.store_id, op.meta.version), op.meta)
          break
        case 'meta_delete':
          metas.delete(meta_key(op.store_id, op.version))
          break
        case 'data_put':
          blobs.set(op.data_key, op.bytes)
          break
        case 'observation_put':
          rows.set(op.row.id, op.row)
          break
        case 'observation_delete':
          rows.delete(op.id)
          break
      }
    }
    return ok(undefined)
  }

  return { metadata, data, observations, on_event: options.on_event, apply_batch }
}
```

Use it like any built-in backend:

```typescript
import { create_corpus, define_store, text_codec } from '@f0rbit/corpus'
import type { Backend } from '@f0rbit/corpus'

declare function create_map_backend(): Backend // from the example above

const notes = create_corpus()
  .with_backend(create_map_backend())
  .with_store(define_store('notes', text_codec()))
  .build()

const put = await notes.stores.notes.put('hello')
```

### Errors are values

Every `Result`-returning method must **return** its failure, never throw it. Don't count on corpus to catch a throw for you: it can reach the caller as a rejected promise instead of a `CorpusError`. Wrap I/O that can reject (network calls, `fetch`, driver queries) at the edge with `try_catch_async` and map the failure to a `CorpusError`, usually `{ kind: 'storage_error', operation, cause }`. Use `not_found` for absent records; it is not a storage error.

Keep error messages about *what* failed (status code, key, SQL error text). Never put credentials or connection strings in an error message.

### `list_stores` (optional)

`metadata.list_stores` enumerates every store id. The `corpus` CLI's `stores` command and an unfiltered `copy()` need it. Callers that need it and find it missing report an actionable error rather than treating the backend as empty, so leaving it out is safe. All built-in backends yield the ids sorted.

### Observations (optional)

`create_observations_storage` and `create_observations_client` are public. They turn a row-level adapter (`get_all`, `set_all`, `get_one`, `add_one`, `remove_one`) into a full `ObservationsClient`: validation, staleness, filtering and `entity_key` inheritance included. Pass your `metadata` client so staleness checks and `entity_key` inheritance read your snapshots. Pass your emitter so rejected queries surface as `validation_error` events.

An adapter can also provide `query(opts)` and `delete_by_source(...)` to push filtering into your storage. Without them, the storage wrapper loads every row and filters in memory. See `ObservationsAdapter` and `StorageQueryOpts` in [Observations](/corpus/api/core/observations).

### `on_event`

Return the handler you were given as `on_event` and call it from your methods. Emit `error` events at minimum: listing methods have no other way to report a failure (see [semantics](#semantics-a-backend-must-honour)). Tools such as `with_error_tracking` and the CLI hook in through `on_event`.

### `apply_batch` (optional)

`corpus.transaction()` buffers its body's ops and submits them as one `BatchOp[]` at commit:

```typescript
type BatchOp =
  | { type: 'meta_put'; meta: SnapshotMeta }
  | { type: 'meta_delete'; store_id: string; version: string }
  | { type: 'data_put'; data_key: string; bytes: Uint8Array }
  | { type: 'observation_put'; row: ObservationRow }
  | { type: 'observation_delete'; id: string }
```

Ops arrive in body order. You may reorder internally, for example writing all blobs before any metadata, as long as the final state matches applying the ops in their original order.

`apply_batch` must be **all-or-nothing from a reader's point of view**:

- On success, return `ok(undefined)`. Every op is visible to later reads.
- On failure, return `err({ kind: 'transaction_aborted', reason: 'apply_batch_failed', cause })`. No op may be visible. A repeated caller-supplied observation id is one of these failures.
- If your storage can't guarantee that (some ops already landed), say so with `partial_commit`:

```typescript
import { err } from '@f0rbit/corpus'
import type { CorpusError, Result } from '@f0rbit/corpus'

const report_partial = (ops_completed: number, ops_failed: number, cause: Error): Result<void, CorpusError> =>
  err({ kind: 'partial_commit', ops_completed, ops_failed, cause })
```

**Leaving `apply_batch` out** is valid. The remote backend does, because D1's HTTP API has no multi-statement transaction. `corpus.transaction()` then applies ops one at a time against `metadata` and `data`. On the first failure it undoes what it can, in reverse order:

- `meta_put` → `metadata.delete`
- `data_put` → `data.delete` (best-effort; data is content-addressed)
- `meta_delete` / `observation_delete` can't be undone, so any that already applied turn the result into `partial_commit`

If every undo succeeds, the transaction returns `transaction_aborted`; if any undo fails, `partial_commit`. Observation ops always fail on this path (`observation ops in transactions require backend.apply_batch`), so **a backend that supports observations inside transactions must implement `apply_batch`**. The fallback is a safety net for simple stores, not a transaction.

<Aside type="tip">
  Native storage transactions (SQL `BEGIN`/`COMMIT`, D1's `db.batch()`) make `apply_batch` straightforward. When blobs and metadata live in different systems, write the blobs first (content-addressed, so orphans are harmless) and commit the metadata last as one atomic step. The Cloudflare backend works this way.
</Aside>

## Wrapping an existing backend

A `Backend` is a plain object, so decorators are object spreads. The example below counts operations and injects a failure into data writes. Both wrap `create_memory_backend()`, and the result goes into `create_layered_backend`:

```typescript
import { create_layered_backend, create_memory_backend, err } from '@f0rbit/corpus'
import type { Backend, CorpusError, Result } from '@f0rbit/corpus'

export function with_metrics(inner: Backend, counts: Map<string, number>): Backend {
  const tick = (op: string): void => {
    counts.set(op, (counts.get(op) ?? 0) + 1)
  }
  return {
    ...inner,
    metadata: {
      ...inner.metadata,
      get: (store_id, version) => {
        tick('meta.get')
        return inner.metadata.get(store_id, version)
      },
      put: (meta) => {
        tick('meta.put')
        return inner.metadata.put(meta)
      },
    },
    data: {
      ...inner.data,
      put: (data_key, bytes) => {
        tick('data.put')
        return inner.data.put(data_key, bytes)
      },
    },
  }
}

// Fails every data write after the first `allowed`. Dropping apply_batch
// forces corpus.transaction() onto the sequential fallback, so compensation
// and partial_commit paths get exercised.
export function with_failing_data_puts(inner: Backend, allowed: number): Backend {
  let puts = 0
  const { apply_batch: _atomic, ...rest } = inner
  return {
    ...rest,
    data: {
      ...inner.data,
      async put(data_key, bytes): Promise<Result<void, CorpusError>> {
        puts += 1
        if (puts > allowed) {
          return err({ kind: 'storage_error', operation: 'data.put', cause: new Error('injected failure') })
        }
        return inner.data.put(data_key, bytes)
      },
    },
  }
}

const counts = new Map<string, number>()
const cache = with_metrics(create_memory_backend(), counts)
const durable = create_memory_backend()

const layered = create_layered_backend({
  read: [cache, durable],
  write: [cache, durable],
})
```

Things to know when you wrap:

- **Spreading keeps `inner.apply_batch`**, and it writes to the inner storage directly. Transactional writes bypass your wrapped `metadata`/`data` methods. Wrap or drop `apply_batch` if the decorator must see every write.
- **`inner.observations` was built over `inner.metadata`**, not your wrapper. Observation reads and staleness checks don't go through the decorator.
- The built-in clients don't depend on `this`, so spreading their methods is safe. If a client you wrap does use `this`, bind it.
- `create_layered_backend` sends `apply_batch` and `list_stores` to the **bottom write layer only**. Transactional writes skip cache layers, which fill lazily on reads. See [Layered](/corpus/api/backends/layered).

## Semantics a backend must honour

Every built-in backend follows these rules, and code written against one backend relies on them:

- **Newest first, ties by insertion order.** `list` yields newest `created_at` first. When two snapshots share a `created_at` (same millisecond), the most recently **put** one comes first, and `get_latest` returns it. Re-putting an existing `(store_id, version)` keeps its original insertion position. Never order by the version string: versions are not lexicographically sortable.
- **Filters apply before `limit`.** `before`, `after`, `tags` (every tag must be present), and `entity_key` (exact match; unkeyed snapshots never match a keyed filter) all narrow the set *before* `limit` truncates it. In SQL, they belong in the `WHERE` clause; filtering in JavaScript after a SQL `LIMIT` returns too few rows. `cursor` is not interpreted by any built-in backend.
- **`get_latest(store_id, { entity_key })`** scopes to that entity. Without `entity_key` it is store-wide and ignores entities. Not found means `err({ kind: 'not_found', store_id, version: 'latest' })`.
- **Listings report failures as `error` events, then end.** `list`, `list_stores` and `get_children` return `AsyncIterable`s, which have no error channel. On a storage failure, emit `{ type: 'error', error }` through `on_event` and finish the iterable. Callers that must tell "empty" from "failed" use `with_error_tracking` / `collect_listing`, which read those events.
- **`data.exists` returns a plain boolean.** On a storage failure, emit an `error` event and return `false`.
- **`find_by_hash` searches the whole history** for deduplication. Change detection (`put(data, { if_changed: true })`) goes through `get_latest` instead.
- **Optional fields are omitted, not `undefined`.** If your storage returns `NULL` for `entity_key`, `invoked_at` or `tags`, drop the key before returning the `SnapshotMeta`.
- **Observation ids are caller-supplied or generated, never overwritten.** Putting an id that already exists returns `validation_error`. Inside `apply_batch` it aborts the batch with `transaction_aborted`. Check for the collision in the same atomic section as the insert.
- **Data before metadata.** Metadata must never point at a blob that doesn't exist. `copy()` and the transaction paths write `data` before `metadata` and rely on it. Orphaned blobs are harmless because they are content-addressed.

## Testing a custom backend

Corpus's own **backend contract suite** and **backend equivalence models** live in its repository tests (`tests/integration/backend-contract.test.ts`, `tests/property/backend-equivalence.test.ts`). They are not a published export. To use them, copy them into your project as a starting point.

What *is* published is `law.provider_equivalence` from `@f0rbit/corpus/testing`. It runs generated command sequences against a small reference model and every provider you list, comparing results after each command, and shrinks any divergence to a minimal sequence. Put `create_memory_backend()` next to your backend: when both agree with the model, your backend behaves like the reference.

```typescript
import { test } from 'bun:test'
import { create_memory_backend, err, ok } from '@f0rbit/corpus'
import type { Backend, CorpusError, ParentRef, Result, SnapshotMeta } from '@f0rbit/corpus'
import { fc, law } from '@f0rbit/corpus/testing'

declare function create_map_backend(): Backend // the backend under test

type Model = { metas: SnapshotMeta[] }

const store_ids = ['a', 'b']

const meta_arb: fc.Arbitrary<SnapshotMeta> = fc.record({
  store_id: fc.constantFrom(...store_ids),
  version: fc.constantFrom('v1', 'v2', 'v3'),
  parents: fc.constant<ParentRef[]>([]),
  created_at: fc.integer({ min: 0, max: 2 }).map((s) => new Date(s * 1000)), // few values → ties
  content_hash: fc.constantFrom('h1', 'h2'),
  content_type: fc.constant('text/plain'),
  size_bytes: fc.nat(100),
  data_key: fc.constantFrom('k1', 'k2'),
})

const in_store = (m: Model, store_id: string): SnapshotMeta[] => m.metas.filter((x) => x.store_id === store_id)

const newest_first = (metas: SnapshotMeta[]): SnapshotMeta[] =>
  metas.toReversed().toSorted((a, b) => b.created_at.getTime() - a.created_at.getTime())

const put = (meta: SnapshotMeta) =>
  law.equivalence_command<Model, Backend, Result<void, CorpusError>>({
    label: `put(${meta.store_id}/${meta.version})`,
    on_model: (m) => {
      const i = m.metas.findIndex((x) => x.store_id === meta.store_id && x.version === meta.version)
      if (i === -1) m.metas.push(meta)
      else m.metas[i] = meta
      return ok(undefined)
    },
    on_sut: (b) => b.metadata.put(meta),
  })

const get = (store_id: string, version: string) =>
  law.equivalence_command<Model, Backend, Result<SnapshotMeta, CorpusError>>({
    label: `get(${store_id}/${version})`,
    on_model: (m) => {
      const found = in_store(m, store_id).find((x) => x.version === version)
      if (found === undefined) return err({ kind: 'not_found', store_id, version })
      return ok(found)
    },
    on_sut: (b) => b.metadata.get(store_id, version),
  })

const latest = (store_id: string) =>
  law.equivalence_command<Model, Backend, Result<SnapshotMeta, CorpusError>>({
    label: `get_latest(${store_id})`,
    on_model: (m) => {
      const found = newest_first(in_store(m, store_id)).at(0)
      if (found === undefined) return err({ kind: 'not_found', store_id, version: 'latest' })
      return ok(found)
    },
    on_sut: (b) => b.metadata.get_latest(store_id),
  })

const list = (store_id: string, limit: number) =>
  law.equivalence_command<Model, Backend, Result<SnapshotMeta[], CorpusError>>({
    label: `list(${store_id}, limit=${String(limit)})`,
    on_model: (m) => ok(newest_first(in_store(m, store_id)).slice(0, limit)),
    on_sut: async (b) => ok(await Array.fromAsync(b.metadata.list(store_id, { limit }))),
  })

test('map backend is equivalent to the memory backend', async () => {
  await law.provider_equivalence<Model, Backend>({
    model: () => ({ metas: [] }),
    providers: [
      { label: 'memory', build: () => create_memory_backend() },
      { label: 'map', build: () => create_map_backend() },
    ],
    commands: [
      meta_arb.map(put),
      fc.tuple(fc.constantFrom(...store_ids), fc.constantFrom('v1', 'v2', 'v3')).map(([s, v]) => get(s, v)),
      fc.constantFrom(...store_ids).map(latest),
      fc.tuple(fc.constantFrom(...store_ids), fc.integer({ min: 1, max: 3 })).map(([s, n]) => list(s, n)),
    ],
    equivalence: { results_agree: (a, b) => JSON.stringify(a) === JSON.stringify(b) },
    numRuns: 200,
    maxCommands: 30,
    size: 'max',
  })
})
```

Grow the command pool to match your backend's surface: `list` with `tags`/`entity_key`/`before`/`after`, `delete`, `data.put`/`data.get`, observations. The model is the spec, so keep it small and obviously correct. Payload arbitraries inside commands must be plain `fc.*` values, not `testing.compose`, because the latter's shrinker may not terminate. See [Testing laws](/corpus/testing/laws/) for the rest of the substrate.

<Aside type="caution">
  The memory backend implements `apply_batch`; a backend without it takes the sequential transaction fallback. Test transactions separately against your backend, including a failing write, which `with_failing_data_puts` above can simulate.
</Aside>
